```python
import csv,io,re,subprocess
from pathlib import Path
base='ce2d8122f91744ddca2a177c6c49b9255e5f9f42'
ledger='docs/roadmap/Kafka_Learning_Lab_Task_Checklist.csv'
def git(*args):return subprocess.check_output(['git',*args],text=True,encoding='utf-8').strip()
def rows(data):return list(csv.reader(io.StringIO(data.decode('utf-8-sig'),newline='')))
old=rows(subprocess.check_output(['git','show',base+':'+ledger]));raw=Path(ledger).read_bytes();new=rows(raw)
assert len(old)==len(new)==727 and old[0]==new[0] and len(new[0])==20
assert [r[0] for r in new]==[r[0] for r in old] and len({r[0] for r in new[1:]})==726 and len({r[1] for r in new[1:]})==15
assert all(len(r)==20 and not any(r[18:]) for r in new)
assert not raw.startswith(b'\xef\xbb\xbf') and raw.count(b'\r\n')==raw.count(b'\n')==727
selected={f'P1-{n:03}' for n in range(1,13)};allowed={'Status','Completed_Date','Notes','Evidence_Link'}
for before,after in zip(old[1:],new[1:]):
    if after[0] not in selected:assert before==after,after[0]
    else:assert {new[0][i] for i in range(20) if before[i]!=after[i]}<=allowed
records={r[0]:dict(zip(new[0],r)) for r in new[1:]}
assert all(records[t]['Status']=='Done' for t in selected)
assert all(r['Status']=='Not Started' for t,r in records.items() if t.startswith('P1-') and t not in selected)
sections=re.findall(r'^## [0-9]+[.] .+$',Path('docs/learning-notes/TEMPLATE.md').read_text(encoding='utf-8'),re.M)
for task in selected:
    note=Path(f'docs/learning-notes/P1/{task}.md');text=note.read_text(encoding='utf-8')
    assert all(h in text for h in sections) and '**Status:** Done' in text
    assert Path(records[task]['Evidence_Link']).is_file()
assert len(list(Path('docs/learning-notes/P1').glob('P1-*.md')))==12
assert git('diff','--name-only',base,'--','docs/roadmap').splitlines()==[ledger]
assert not Path('applications').exists() and not Path('infrastructure').exists() and not Path('pom.xml').exists()
assert not list(Path('.github/workflows').glob('*.yml')) and not list(Path('.github/workflows').glob('*.yaml'))
changed=set(git('diff','--name-only',base).splitlines())|set(git('ls-files','--others','--exclude-standard').splitlines())
linkcount=0
for name in sorted(changed):
    if not name.endswith('.md'):continue
    path=Path(name);text=re.sub(r'```.*?```','',path.read_text(encoding='utf-8-sig'),flags=re.S)
    for target in re.findall(r'\[[^\]]+\]\(([^)]+)\)',text):
        if target.startswith(('https://','http://','#','mailto:')):continue
        assert (path.parent/target.split('#')[0]).exists(),(name,target)
        linkcount+=1
remote=dict((ref,sha) for sha,ref in (line.split() for line in git('ls-remote','--heads','origin').splitlines()))
branches=git('for-each-ref','--format=%(refname:short) %(objectname)','refs/heads/').splitlines();count=0
for line in branches:
    branch,sha=line.split();m=re.search(r'/P1-([0-9]{3})-',branch)
    if m and 1<=int(m.group(1))<=11:
        assert remote.get('refs/heads/'+branch)==sha,branch
        task=f'P1-{int(m.group(1)):03}'
        files=git('show','--pretty=','--name-only',sha).splitlines()
        assert f'docs/learning-notes/P1/{task}.md' in files and ledger in files
        priorrows=rows(subprocess.check_output(['git','show',sha+'^:'+ledger]));commitrows=rows(subprocess.check_output(['git','show',sha+':'+ledger]))
        assert [b[0] for a,b in zip(priorrows,commitrows) if a!=b]==[task]
        count+=1
assert count==11
subprocess.run(['git','diff','--check',base],check=True)
print('PASS: 12 Done P1 tasks, 12 complete learning notes/evidence; P1-013 through P1-043 remain Not Started')
print('PASS: 726 IDs/order, 15 phases, header/20 physical columns, UTF-8/CRLF and every unrelated CSV row preserved')
print(f'PASS: {len(changed)} scoped changed files and {linkcount} local prose links; original roadmap baseline preserved')
print('PASS: 11 preceding task branches match GitHub; every preceding commit changes exactly its own CSV row')
print('PASS: no application/infrastructure/build/CI implementation; git diff --check passed')
print('P1-012 commit/push and remote-tip verification remain the final publication barrier')
```
